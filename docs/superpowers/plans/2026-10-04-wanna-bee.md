# Wanna Bee Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a fourth trading style, "Contrarian Momentum" (`wanna`), and swap the slot-2 bee for "Wanna Bee" running on it, without touching slots 1 and 3.

**Architecture:** A new brain `src/bees/wanna.ts` offers Jev funding-gated longs on rising-volume 7-day leaders and shorts on rising-volume laggards. A new optional `BeeBrain.forcedExit` hook, evaluated by the pure risk layer right after the code stop, enforces the owner's code-side exits (funding flip, half off at +6 %). A small `src/swap.ts` + CLI replaces one slot in `settings.json` and clears that slot's rows from the trade DB.

**Tech Stack:** TypeScript (ESM, `.js` import suffixes), Node 22+ `node:sqlite`, zod, vitest, pnpm, Docker Compose.

**Spec:** `docs/superpowers/specs/2026-10-04-wanna-bee-design.md`

## Global Constraints

- Branch `wanna-bee`. Commit after every task. Commit messages end with:

- Existing styles (`bizzy`, `breezy`, `boozy`) must behave byte-for-byte as before; brains without `forcedExit` are unaffected.
- Funding units: `CoinStats.fundingPct` is **percent per 8h** (OKX rate × 100). Entry edge 0.002, exit on bare sign flip.
- Stop `clamp(4 + 2 × atr1hPct, 6, 8)` %, 7 % when ATR unknown; `atr1hPct = 2 × atr14Pct` (15m ATR, as in Momentum).
- Take half once at mark ≥ +6 % from average entry. Trail after half: `max(entry, mark − 3 × ATR1h)` (long; mirrored short).
- Size 0.5 of max notional (≈ 1 × equity). Knobs `WANNA_*`: trades 6, fee $3.00, spread 15 bp, cooldown 5 min, stop-ATR 2, max-flat 0.

- Code style: match surrounding files (2-space indent, double quotes, one-line JSDoc on exported helpers, no new dependencies).

## Review Focus

1. **Held coin's funding is unknown** (funding fetch failed, `fundingPct: null`): no funding-flip exit, no crash; take-half still works. → test in Task 2.
2. **Held coin missing from market stats** (data failure for that coin): `forcedExit` and `trail` return null; the menu still offers HOLD/CLOSE. → test in Task 2.
3. **Half-trim rounds to zero contracts** (tiny position): the engine must still mark `halfTaken`, or the forced trim fires every tick forever. → engine test in Task 1.
4. **Funding flips while the bee is benched** (trade cap / fee budget used up) or **Jev is down / over its daily cap**: the exit must still fire. → risk tests in Task 1.
5. **Swapping a slot that holds an open position**: refuse unless `--force`, and never touch other slots' rows. → test in Task 3.

```ts
// See strategies/WANNA_BEE.md.
import type { CoinStats, MarketView } from "../market/types.js";
import { zs } from "./boozy.js";
import { r2 } from "./common.js";
import type { BeeBrain, BeeContext, ForcedExit, Menu, Side } from "./types.js";

/** Funding (% per 8h) must favour the side by at least this much to enter; the exit is the bare sign flip. */
export const WANNA_FUNDING_EDGE_PCT = 0.002;
/** Take half off once the mark is this far (%) past the average entry. */
export const WANNA_TAKE_HALF_PCT = 6;
/** Enter at 1x equity (half of the 2x max). No adds. */
const ENTRY_FRAC = 0.5;
/** Leaders = top 20% of the ranked pool, laggards = bottom 20%, at least 5 coins each. */
const BUCKET = 0.2;
const BUCKET_MIN = 5;
const PER_SIDE = 4;
/** After the half: trail 3 x ATR(1h) behind the mark, never below break-even. */
const TRAIL_ATR1H = 3;

/** 1h ATR in % of price, approximated as 2 x the 15m ATR (as Momentum does). */
const atr1hPct = (s: CoinStats | undefined) => (s && s.atr14Pct !== null ? s.atr14Pct * 2 : null);

/** Stop distance in %: 7 at a typical 1.5% hourly swing, 6 on calm coins, 8 on jumpy ones, 7 when unknown. */
export function wannaStopPct(s: CoinStats | undefined): number {
  const a = atr1hPct(s);
  if (a === null || !Number.isFinite(a)) return 7;
  return Math.min(8, Math.max(6, 4 + 2 * a));
}

export interface WannaEntry {
  s: CoinStats;
  side: Side;
  score: number;
}

/** Eligible longs (best first) and shorts (weakest first): rising volume, funding on the bee's side. */
export function wannaEntries(view: MarketView, spreadGateBps: number): { longs: WannaEntry[]; shorts: WannaEntry[] } {
  const pool = view.gated
    .map((id) => view.stats.get(id))
    .filter((s): s is CoinStats => !!s && s.ret7dPct !== null && s.ret24hPct !== null && s.spreadBp <= spreadGateBps);
  if (!pool.length) return { longs: [], shorts: [] };
  const z7 = zs(pool.map((s) => s.ret7dPct!));
  const z24 = zs(pool.map((s) => s.ret24hPct!));
  const ranked = pool.map((s) => ({ s, score: z7(s.ret7dPct!) + 0.1 * z24(s.ret24hPct!) })).sort((a, b) => b.score - a.score);
  const n = Math.max(BUCKET_MIN, Math.ceil(ranked.length * BUCKET));
  const rising = (s: CoinStats) => s.volZ !== null && s.volZ > 0;
  const longs = ranked
    .slice(0, n)
    .filter((c) => rising(c.s) && c.s.fundingPct !== null && c.s.fundingPct <= -WANNA_FUNDING_EDGE_PCT)
    .slice(0, PER_SIDE)
    .map((c) => ({ ...c, side: "long" as const }));
  const shorts = ranked
    .slice(-n)
    .reverse()
    .filter((c) => rising(c.s) && c.s.fundingPct !== null && c.s.fundingPct >= WANNA_FUNDING_EDGE_PCT)
    .slice(0, PER_SIDE)
    .map((c) => ({ ...c, side: "short" as const }));
  return { longs, shorts };
}

const entries = (ctx: BeeContext) => {
  const { longs, shorts } = wannaEntries(ctx.view, ctx.knobs.spreadGateBps);
  return [...longs, ...shorts];
};

export const wanna: BeeBrain = {
  id: "wanna",
  neverForce: true,
  strategy:
    "You are wanna-bee, the contrarian. Momentum with a funding edge: go long the week's strongest coins while their volume is rising and shorts are paying funding, go short the weakest while their volume is rising and longs are paying. Every coin on your menu already has funding on its side; prefer the one where funding is most stretched against the crowd (fund_pct, fund_z). No holding period and no loyalty: CLOSE or switch the moment a better leader or laggard shows up. The code sets a 6-8% stop, takes half off at +6%, trails the rest, and closes the moment funding flips against you.",
  convictionLabels: ["meh", "keen", "eager", "all-in"],

  universe(ctx) {
    return entries(ctx).map((e) => e.s.instId);
  },

  snapshotCoins(ctx) {
    const ids = entries(ctx).map((e) => e.s.instId);
    const p = ctx.bee.position;
    if (p && !ids.includes(p.instId)) ids.push(p.instId);
    return ids;
  },

  coinSnapshot(s) {
    return {
      r1h_pct: r2(s.ret1hPct, 1),
      r24h_pct: r2(s.ret24hPct, 0),
      r7d_pct: r2(s.ret7dPct, 0),
      vol_z: r2(s.volZ, 1),
      fund_pct: r2(s.fundingPct, 4),
      fund_z: r2(s.fundingZ, 1),
      spread_bp: r2(s.spreadBp, 0),
      vol_musd: r2(s.vol24hUsd / 1e6, 1),
    };
  },

  menu(ctx) {
    const p = ctx.bee.position;
    const m: Menu = {};
    if (p) {
      m.HOLD = { desc: "keep position", intent: { kind: "hold" } };
      m.CLOSE = { desc: "close now", intent: { kind: "close", reason: "close" } };
    }
    for (const e of entries(ctx)) {
      if (p && e.s.instId === p.instId) continue;
      const label = `${e.side.toUpperCase()}_${e.s.coin}`;
      m[label] = p
        ? { desc: `close ${p.coin}, ${e.side} ${e.s.coin}`, intent: { kind: "switch", instId: e.s.instId, side: e.side, sizeFrac: ENTRY_FRAC, setup: "strict" } }
        : { desc: null, intent: { kind: "open", instId: e.s.instId, side: e.side, sizeFrac: ENTRY_FRAC, setup: "strict" } };
    }
    return m;
  },

  forcedEntry() {
    return null;
  },

  sizeFrac(intent) {
    // Always 1x equity; the size never depends on conviction.
    return intent.sizeFrac;
  },

  stopFor(instId, side, entryPx, ctx) {
    const pct = wannaStopPct(ctx.view.stats.get(instId)) / 100;
    return side === "long" ? entryPx * (1 - pct) : entryPx * (1 + pct);
  },

  trail(ctx) {
    const p = ctx.bee.position;
    const s = p ? ctx.view.stats.get(p.instId) : undefined;
    if (!p || !p.halfTaken || !s) return null;
    const a = atr1hPct(s);
    const dist = a === null ? 0 : (s.mid * a * TRAIL_ATR1H) / 100;
    return p.side === "long" ? Math.max(p.entryPx, s.mid - dist) : Math.min(p.entryPx, s.mid + dist);
  },

  forcedExit(ctx): ForcedExit | null {
    const p = ctx.bee.position;
    const s = p ? ctx.view.stats.get(p.instId) : undefined;
    if (!p || !s) return null;
    const f = s.fundingPct;
    if (f !== null && (p.side === "long" ? f > 0 : f < 0)) {
      return { intent: { kind: "close", reason: "funding_flip" }, reason: "funding_flip", status: `funding flipped on ${p.coin} (${f.toFixed(4)}%): closed` };
    }
    const gainPct = p.entryPx > 0 ? ((p.side === "long" ? 1 : -1) * (s.mid - p.entryPx) * 100) / p.entryPx : 0;
    if (!p.halfTaken && gainPct >= WANNA_TAKE_HALF_PCT) {
      return { intent: { kind: "trim", fraction: 0.5 }, reason: "take_half", status: `took half off ${p.coin} at +${gainPct.toFixed(1)}%` };
    }
    return null;
  },

  idleStatus() {
    return "waiting: no leader or laggard with funding on its side";
  },
};
```

Note: `trail` with an unknown ATR returns the break-even/mark floor (`dist = 0` → `max(entry, mark)`), i.e. it locks in the current mark; acceptable because the ATR is only unknown with no 15m candles.

```ts
// Swap one bee without running Setup again: back up, replace that slot in settings.json, clear the slot's rows in the
// trade DB (it restarts at the start equity on the next engine start). Run with the engine stopped.
import { cpSync, existsSync, mkdirSync, rmSync } from "node:fs";
import { basename, dirname, join } from "node:path";
import { DatabaseSync } from "node:sqlite";
import type { BeeId } from "./config.js";
import { BEES } from "./config.js";
import { BeeSchema, loadSettings, saveSettings, SettingsSchema, type BeeSettings } from "./settings.js";
import { imageDir } from "./setup.js";

/** Every table with a per-bee row. */
export const SLOT_TABLES = ["bee_state", "decisions", "orders", "fills", "funding", "equity_snapshots", "reconciliations", "caps"] as const;

export interface SwapOptions {
  settingsPath: string;
  dbPath: string;
  slot: BeeId;
  bee: BeeSettings;
  now: number;
  /** Swap even if the slot holds a position (it is dropped from the books, not closed on an exchange). */
  force?: boolean;
}

export function swapSlot(o: SwapOptions): { backupDir: string } {
  const bee = BeeSchema.parse(o.bee);
  const settings = loadSettings(o.settingsPath);
  if (!settings) throw new Error(`${o.settingsPath} not found: run Setup first`);
  const i = BEES.indexOf(o.slot);
  const next = SettingsSchema.parse({ ...settings, bees: settings.bees.map((b, j) => (j === i ? bee : b)) });

  const db = new DatabaseSync(o.dbPath);
  try {
    const row = db.prepare(`SELECT json FROM bee_state WHERE bee = ?`).get(o.slot) as { json: string } | undefined;
    const open = row ? (JSON.parse(row.json) as { position: unknown }).position : null;
    if (open && !o.force) throw new Error(`${o.slot} has an open position; close it first or pass --force`);

    const backupDir = join(dirname(o.settingsPath), `pre-swap-${o.slot}-${o.now}`);
    mkdirSync(backupDir, { recursive: true });
    cpSync(o.settingsPath, join(backupDir, basename(o.settingsPath)));
    db.exec(`VACUUM INTO '${join(backupDir, basename(o.dbPath)).replaceAll("'", "''")}'`);
    const images = imageDir(o.settingsPath);
    if (existsSync(images)) cpSync(images, join(backupDir, "bee-images"), { recursive: true });

    db.exec("BEGIN");
    for (const t of SLOT_TABLES) db.prepare(`DELETE FROM ${t} WHERE bee = ?`).run(o.slot);
    db.exec("COMMIT");
    rmSync(join(images, `${o.slot}.jpg`), { force: true });
    saveSettings(o.settingsPath, next);
    return { backupDir };
  } finally {
    db.close();
  }
}
```
